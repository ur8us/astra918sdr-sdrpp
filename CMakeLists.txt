cmake_minimum_required(VERSION 3.20)
project(astra918_sdrpp_source VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ASTRA918_BUILD_MODULE "Build external SDR++ module" OFF)
option(ASTRA918_SANITIZERS "Address/undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND AND NOT CMAKE_CROSSCOMPILING)
  pkg_check_modules(LIBUSB QUIET IMPORTED_TARGET libusb-1.0)
endif()
if(TARGET PkgConfig::LIBUSB)
  set(ASTRA_USB PkgConfig::LIBUSB)
else()
  find_path(LIBUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0 REQUIRED)
  find_library(LIBUSB_LIBRARY NAMES usb-1.0 libusb-1.0 REQUIRED)
  add_library(astra_usb UNKNOWN IMPORTED)
  set_target_properties(astra_usb PROPERTIES IMPORTED_LOCATION "${LIBUSB_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${LIBUSB_INCLUDE_DIR}")
  set(ASTRA_USB astra_usb)
endif()
add_library(astra918 STATIC src/cmx918_protocol.cpp src/cmx918_device.cpp src/usb_transport.cpp src/tcp_transport.cpp)
target_include_directories(astra918 PUBLIC src)
target_link_libraries(astra918 PUBLIC ${ASTRA_USB} Threads::Threads)
if(WIN32)
  target_link_libraries(astra918 PUBLIC ws2_32)
endif()
if(MSVC)
  target_compile_options(astra918 PRIVATE /W4)
else()
  target_compile_options(astra918 PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ASTRA918_SANITIZERS AND NOT MSVC)
  target_compile_options(astra918 PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(astra918 PUBLIC -fsanitize=address,undefined)
endif()
add_executable(astra918_tests tests/astra_tests.cpp)
target_link_libraries(astra918_tests PRIVATE astra918)
enable_testing()
add_test(NAME protocol_stream COMMAND astra918_tests)
add_executable(astra918_sim_test tests/simulator.cpp)
target_link_libraries(astra918_sim_test PRIVATE astra918)
if(ASTRA918_BUILD_MODULE)
  if(NOT SDRPP_SOURCE_DIR)
    message(FATAL_ERROR "Set SDRPP_SOURCE_DIR to the matching SDR++ source checkout")
  endif()
  if(WIN32 AND NOT SDRPP_CORE_LIBRARY)
    message(FATAL_ERROR "Windows module builds require matching SDRPP_CORE_LIBRARY import library")
  endif()
  get_filename_component(SDRPP_SOURCE_DIR "${SDRPP_SOURCE_DIR}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
  if(SDRPP_CORE_LIBRARY)
    get_filename_component(SDRPP_CORE_LIBRARY "${SDRPP_CORE_LIBRARY}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
  add_library(astra918_source SHARED src/sdrpp_source_module.cpp)
  if(MSVC)
    target_compile_options(astra918_source PRIVATE /utf-8)
  endif()
  set_target_properties(astra918_source PROPERTIES PREFIX "")
  target_include_directories(astra918_source PRIVATE "${SDRPP_SOURCE_DIR}/core/src" "${SDRPP_SOURCE_DIR}/core/src/imgui")
  # External SDK include/library paths, especially non-system macOS/Windows prefixes.
  target_include_directories(astra918_source PRIVATE ${SDRPP_SDK_INCLUDE_DIRS})
  target_link_directories(astra918_source PRIVATE ${SDRPP_SDK_LIBRARY_DIRS})
  target_link_libraries(astra918_source PRIVATE ${SDRPP_SDK_LIBRARIES})
  if(PkgConfig_FOUND)
    pkg_check_modules(SDRPP_DEPS QUIET IMPORTED_TARGET volk fftw3f glfw3)
    if(TARGET PkgConfig::SDRPP_DEPS)
      target_link_libraries(astra918_source PRIVATE PkgConfig::SDRPP_DEPS)
    endif()
  endif()
  target_link_libraries(astra918_source PRIVATE astra918)
  if(SDRPP_CORE_LIBRARY)
    if(WIN32)
      get_filename_component(SDRPP_CORE_LIBRARY_DIR "${SDRPP_CORE_LIBRARY}" DIRECTORY)
      get_filename_component(SDRPP_CORE_LIBRARY_NAME "${SDRPP_CORE_LIBRARY}" NAME_WE)
      add_library(astra918_sdrpp_core SHARED IMPORTED GLOBAL)
      set_target_properties(astra918_sdrpp_core PROPERTIES
        IMPORTED_IMPLIB "${SDRPP_CORE_LIBRARY}"
        IMPORTED_LOCATION "${SDRPP_CORE_LIBRARY_DIR}/${SDRPP_CORE_LIBRARY_NAME}.dll")
      target_link_libraries(astra918_source PRIVATE astra918_sdrpp_core)
    else()
      target_link_libraries(astra918_source PRIVATE "${SDRPP_CORE_LIBRARY}")
    endif()
  elseif(APPLE)
    # SDR++ resolves core symbols from the hosting executable when it loads a
    # third-party module. macOS requires this explicit linker mode.
    target_link_options(astra918_source PRIVATE "LINKER:-undefined,dynamic_lookup")
  endif()
  add_executable(astra918_module_test tests/module_simulator.cpp)
  target_link_libraries(astra918_module_test PRIVATE ${CMAKE_DL_LIBS})
  if(ASTRA918_SANITIZERS AND NOT MSVC)
    target_compile_options(astra918_module_test PRIVATE -fsanitize=address,undefined)
    target_link_options(astra918_module_test PRIVATE -fsanitize=address,undefined)
  endif()
  install(TARGETS astra918_source LIBRARY DESTINATION lib/sdrpp/plugins RUNTIME DESTINATION lib/sdrpp/plugins)
endif()
